# F-1 Whole-Foods Seed Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Seed ~240 curated generic whole foods (raw/dry, per-100 g, bilingual) into the shared ingredient library so unlabeled foods are searchable and usable in recipes with correct macros.

**Architecture:** One additive `name_en` column on `ingredients` (mirrors the `exercises` bilingual pattern) + a guarded system seed. The seed SQL is generated by a build script from a curated `foods.json` joined to the public-domain USDA SR Legacy dataset; provenance (FDC id) lives in the data file, not the DB. Two migrations: schema first (unblocks generated types), seed second.

**Tech Stack:** Supabase/Postgres, TypeScript, Vitest, React + react-i18next, USDA FoodData Central SR Legacy (REST API, free api.data.gov key), tsx.

**Spec:** `docs/superpowers/specs/2026-05-23-whole-foods-seed-design.md`

```ts
import { readFileSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';

export interface FoodEntry {
  query: string; // FDC search term (hand-authored)
  name_es: string;
  name_en: string;
  category: string;
  fdc_id?: number; // filled by resolve.ts, reviewed before build
  fdc_description?: string; // matched USDA entry text (provenance/review aid)
}
export interface SRFood {
  fdcId: number;
  foodNutrients: { nutrient: { number: string }; amount?: number }[];
}

// USDA SR nutrient numbers (per 100 g). See plan Task 6 header.
const N = {
  kcal: '208',
  protein: '203',
  fat: '204',
  carbs: '205',
  fiber: '291',
  sugar: '269',
  satfat: '606',
} as const;

function round2(n: number): number {
  return Math.round(n * 100) / 100;
}

export function amountFor(food: SRFood, number: string): number | null {
  const fn = food.foodNutrients.find((x) => x.nutrient.number === number);
  return fn?.amount != null ? round2(fn.amount) : null;
}

export function buildRow(entry: FoodEntry, food: SRFood): string {
  const kcal = amountFor(food, N.kcal);
  if (kcal == null) {
    throw new Error(`fdc ${entry.fdc_id} (${entry.name_es}) has no energy value`);
  }
  // U-1: missing sub-macros stay null (never 0 — that would assert "sugar-free").
  const sugar = amountFor(food, N.sugar);
  const satfat = amountFor(food, N.satfat);
  const num = (v: number | null) => (v == null ? 'null' : String(v));
  const esc = (s: string) => s.replace(/'/g, "''");
  return (
    `      ('${esc(entry.name_es)}', '${esc(entry.name_en)}', 'gram', ` +
    `${kcal}, ${amountFor(food, N.protein) ?? 0}, ${amountFor(food, N.carbs) ?? 0}, ` +
    `${amountFor(food, N.fat) ?? 0}, ${amountFor(food, N.fiber) ?? 0}, ` +
    `${num(sugar)}, ${num(satfat)}, true, null, 'system')`
  );
}

const MIGRATION_HEADER = `-- F-1 step 2/2 — curated whole-foods system seed (~240 generic basics).
-- Generated by scripts/whole-foods/build-seed.ts from scripts/whole-foods/foods.json
-- joined to the public-domain USDA SR Legacy dataset. DO NOT hand-edit — re-run
-- the build. Guarded: re-applying is a no-op once any source='system' ingredient
-- exists. STAGED — applied to prod at a checkpoint.

do $$
begin
  if not exists (select 1 from public.ingredients where source = 'system') then
    insert into public.ingredients
      (name, name_en, unit_type, kcal_per_unit, protein_g_per_unit,
       carbs_g_per_unit, fat_g_per_unit, fiber_g_per_unit,
       sugar_g_per_unit, saturated_fat_g_per_unit,
       is_verified, created_by_user_id, source)
    values
`;

const MIGRATION_FOOTER = `;
  end if;
end $$;
`;

const FDC = 'https://api.nal.usda.gov/fdc/v1';

// Batched detail fetch (FDC allows up to 20 ids per /foods call). Returns the
// SRFood-shaped objects buildRow/amountFor already consume.
async function fetchFoods(ids: number[], key: string): Promise<Map<number, SRFood>> {
  const byId = new Map<number, SRFood>();
  for (let i = 0; i < ids.length; i += 20) {
    const batch = ids.slice(i, i + 20);
    const res = await fetch(`${FDC}/foods?api_key=${key}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ fdcIds: batch, format: 'full' }),
    });
    if (!res.ok) throw new Error(`FDC /foods failed: ${res.status}`);
    const foods = (await res.json()) as SRFood[];
    for (const f of foods) byId.set(f.fdcId, f);
  }
  return byId;
}

async function main(): Promise<void> {
  const dir = resolve(import.meta.dirname);
  const foods = JSON.parse(readFileSync(resolve(dir, 'foods.json'), 'utf8')) as FoodEntry[];
  const key = process.env.FDC_API_KEY;
  if (!key) throw new Error('set FDC_API_KEY (free at https://api.data.gov/signup)');

  const unresolved = foods.filter((f) => f.fdc_id == null);
  if (unresolved.length) {
    throw new Error(`run resolve.ts first — ${unresolved.length} entries lack fdc_id`);
  }

  const ids = foods.map((f) => f.fdc_id!);
  const byId = await fetchFoods(ids, key);
  const missing = ids.filter((id) => !byId.has(id));
  if (missing.length) throw new Error(`FDC returned no food for ids: ${missing.join(', ')}`);

  const rows = foods.map((f) => buildRow(f, byId.get(f.fdc_id!)!));
  const sql = MIGRATION_HEADER + rows.join(',\n') + MIGRATION_FOOTER;
  const out = resolve(dir, '../../supabase/migrations/20260523120100_f1_whole_foods_seed.sql');
  writeFileSync(out, sql);
  console.log(`wrote ${rows.length} rows -> ${out}`);
}

// Run main() only when invoked directly, never on import (keeps the test pure).
if (process.argv[1] && import.meta.url === `file://${process.argv[1]}`) {
  void main();
}
```

- [ ] **Step 3b: Implement `resolve.ts` (curation aid — fills `fdc_id`)**

Create `scripts/whole-foods/resolve.ts`:

```ts
import { readFileSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
import type { FoodEntry } from './build-seed';

const FDC = 'https://api.nal.usda.gov/fdc/v1';

async function searchTop(query: string, key: string): Promise<{ fdcId: number; description: string }> {
  const res = await fetch(`${FDC}/foods/search?api_key=${key}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ query, dataType: ['SR Legacy'], pageSize: 1 }),
  });
  if (!res.ok) throw new Error(`FDC search failed for "${query}": ${res.status}`);
  const json = (await res.json()) as { foods?: { fdcId: number; description: string }[] };
  const top = json.foods?.[0];
  if (!top) throw new Error(`no SR Legacy match for "${query}"`);
  return { fdcId: top.fdcId, description: top.description };
}

async function main(): Promise<void> {
  const dir = resolve(import.meta.dirname);
  const path = resolve(dir, 'foods.json');
  const foods = JSON.parse(readFileSync(path, 'utf8')) as FoodEntry[];
  const key = process.env.FDC_API_KEY;
  if (!key) throw new Error('set FDC_API_KEY (free at https://api.data.gov/signup)');

  let filled = 0;
  for (const f of foods) {
    if (f.fdc_id != null) continue; // already resolved — skip
    const { fdcId, description } = await searchTop(f.query, key);
    f.fdc_id = fdcId;
    f.fdc_description = description;
    filled++;
    console.log(`${f.query} -> ${fdcId}  ${description}`);
  }
  writeFileSync(path, JSON.stringify(foods, null, 2) + '\n');
  console.log(`resolved ${filled} entries; review fdc_description values before building`);
}

if (process.argv[1] && import.meta.url === `file://${process.argv[1]}`) {
  void main();
}
```
